    <script>
        const API_URL = "http://127.0.0.1:5000/api";
        let isLoggedIn = false;
        let currentUser = null;
        let SCRIPTS = []; 

        // --- Core Functions ---

        async function checkAuth() {
            try {
                const res = await fetch(`${API_URL}/me`, { headers: {'Content-Type': 'application/json'}, credentials: 'include' });
                const data = await res.json();
                
                if (data.is_logged_in) {
                    isLoggedIn = true;
                    currentUser = data.username;
                    updateAuthUI(data);
                } else {
                    isLoggedIn = false;
                    updateAuthUI(null);
                }
            } catch (err) { console.error("API Error", err); }
        }

        function updateAuthUI(user) {
            const loginBtn = document.getElementById('nav-login-btn');
            const userControls = document.getElementById('nav-user-controls');
            const mobileLoginBtn = document.getElementById('mobile-login-btn');
            const mobileUserControls = document.getElementById('mobile-user-controls');

            if (isLoggedIn) {
                loginBtn.classList.add('hidden');
                userControls.classList.remove('hidden');
                userControls.classList.add('flex');
                
                document.getElementById('nav-avatar-img').src = user.avatar;
                document.getElementById('profile-avatar-img').src = user.avatar;
                document.getElementById('mobile-avatar-img').src = user.avatar;
                document.getElementById('profile-username').innerText = user.username;
                document.getElementById('mobile-username').innerText = user.username;

                if(mobileLoginBtn) mobileLoginBtn.classList.add('hidden');
                if(mobileUserControls) {
                    mobileUserControls.classList.remove('hidden');
                    mobileUserControls.classList.add('flex');
                }
            } else {
                loginBtn.classList.remove('hidden');
                userControls.classList.add('hidden');
                userControls.classList.remove('flex');

                if(mobileLoginBtn) mobileLoginBtn.classList.remove('hidden');
                if(mobileUserControls) {
                    mobileUserControls.classList.add('hidden');
                    mobileUserControls.classList.remove('flex');
                }
            }
        }

        async function handleLogin(e) {
            e.preventDefault();
            const btn = document.getElementById('btn-login-submit');
            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Authenticating...';
            
            const inputs = e.target.querySelectorAll('input');
            const login = inputs[0].value;
            const password = inputs[1].value;

            try {
                const res = await fetch(`${API_URL}/login`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    credentials: 'include',
                    body: JSON.stringify({ login, password })
                });
                const data = await res.json();

                if (res.ok) {
                    isLoggedIn = true;
                    updateAuthUI(data);
                    showToast("Successfully Logged In!");
                    showPage('home');
                } else {
                    showToast(data.error);
                }
            } catch (err) { showToast("Server Error"); }
            btn.innerHTML = "Sign In";
        }

        async function handleSignup(e) {
            e.preventDefault();
            const btn = document.getElementById('btn-signup-submit');
            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Creating...';
            
            const inputs = e.target.querySelectorAll('input');
            const username = inputs[0].value;
            const email = inputs[1].value;
            const password = inputs[2].value;

            try {
                const res = await fetch(`${API_URL}/signup`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ username, email, password })
                });
                
                if (res.ok) {
                    showToast("Account Created! Please Log In.");
                    showPage('login');
                } else {
                    const data = await res.json();
                    showToast(data.error);
                }
            } catch (err) { showToast("Server Error"); }
            btn.innerHTML = "Create Account";
        }

        }

        function previewFile(input) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    document.getElementById('prev-img').src = e.target.result;
                }
                reader.readAsDataURL(input.files[0]);
            }
        }

        async function handlePostScript(e) {
            e.preventDefault();
            if(!isLoggedIn) return showToast("Please log in first!");

            const btn = e.submitter || e.target.querySelector('button[type="submit"]');
            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Publishing...';

            const title = document.getElementById('post-title').value;
            const game = document.getElementById('post-game').value;
            const desc = document.getElementById('post-desc').value;
            const code = document.getElementById('post-code').value;
            const isKey = document.getElementById('post-key').value === "true";

            // Process Image File
            const fileInput = document.getElementById('post-file');
            let imageData = null;
            if (fileInput.files && fileInput.files[0]) {
                const reader = new FileReader();
                imageData = await new Promise(resolve => {
                    reader.onload = ev => resolve(ev.target.result);
                    reader.readAsDataURL(fileInput.files[0]);
                });
            }

            try {
                const res = await fetch(`${API_URL}/scripts`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    credentials: 'include',
                    body: JSON.stringify({ 
                        title, game, code, description: desc, isKeySystem: isKey, image: imageData 
                    })
                });

                if (res.ok) {
                    showToast("Script Submitted!");
                    fetchScripts();
                    showPage('home');
                } else { showToast("Error publishing script"); }
            } catch (err) { showToast("Connection Error"); }
            
            btn.innerHTML = "Publish Script";
        }

        function showPage(pageId) {

        function highlightCode(code) {
             return code
                .replace(/loadstring/g, '<span class="syntax-func">loadstring</span>')
                .replace(/game/g, '<span class="syntax-var">game</span>')
                .replace(/HttpGet/g, '<span class="syntax-var">HttpGet</span>')
                .replace(/"(.*?)"/g, '<span class="syntax-str">"$1"</span>');
        }

        function renderScripts(query = '') {
            const grid = document.getElementById('scripts-grid');
            const filteredScripts = SCRIPTS.filter(script => {
                const search = query.toLowerCase();
                return script.title.toLowerCase().includes(search) || 
                       script.game.toLowerCase().includes(search) ||
                       script.author.toLowerCase().includes(search);
            });

            if (filteredScripts.length === 0) {
                grid.innerHTML = `<div class="col-span-full text-center py-12 text-gray-500">No scripts found.</div>`;
                return;
            }

            const content = filteredScripts.map(script => `
                <div class="bg-card rounded-xl overflow-hidden group hover:scale-[1.02] transition-all duration-300 shadow-xl border border-white/5 cursor-pointer" onclick="openScriptDetail(${script.id})">
                    <div class="relative h-48 overflow-hidden">
                        <img src="${script.image}" class="w-full h-full object-cover group-hover:brightness-110 transition duration-500">
                        <div class="absolute top-2 left-2 bg-black/60 backdrop-blur-sm px-2 py-1 rounded text-[10px] text-white flex items-center gap-1 font-medium">
                            <i class="fa-regular fa-eye"></i> ${script.views}
                        </div>
                        <div class="absolute top-2 right-2 bg-black/60 backdrop-blur-sm px-2 py-1 rounded text-[10px] text-white font-medium">
                            ${script.date}
                        </div>
                        ${script.isKeySystem ? `<span class="absolute bottom-2 left-2 bg-brand-yellow text-black text-[10px] font-bold px-2 py-1 rounded">Key System</span>` : ''}
                        <div class="absolute bottom-2 right-2 bg-black/60 backdrop-blur-sm px-2 py-1 rounded text-[10px] text-white flex items-center gap-1 font-medium border border-white/10">
                            <i class="fa-solid fa-lock-open"></i> ${script.type}
                        </div>
                    </div>
                    <div class="p-3 bg-card-inner">
                        <div class="text-neon-blue text-xs font-medium truncate mb-1 hover:underline">${script.game}</div>
                        <h3 class="text-white font-bold text-sm leading-tight line-clamp-2 group-hover:text-neon-blue transition-colors">${script.title}</h3>
                    </div>
                </div>
            `).join('');
            
            grid.innerHTML = content;
            const profileGrid = document.getElementById('profile-scripts-grid');
            if(profileGrid && currentUser) {
                const myScripts = SCRIPTS.filter(s => s.author === currentUser);
                profileGrid.innerHTML = myScripts.length ? content : "<div class='text-gray-500'>No saved scripts.</div>";
            }
        }

        function openScriptDetail(id) {
            const script = SCRIPTS.find(s => s.id === id);
            if(!script) return;

            document.getElementById('detail-title').innerText = script.title;
            document.getElementById('detail-author').innerText = '@' + script.author;

            document.getElementById('detail-date').innerText = script.date;
            document.getElementById('detail-image').src = script.image;
            document.getElementById('detail-desc').innerHTML = script.description || "No description provided.";
            document.getElementById('detail-code').innerHTML = highlightCode(script.code);

                badge.className = "bg-green-500 text-black px-3 py-1 rounded text-xs font-bold";
            }
            showPage('script-detail');
        }
